cmake_minimum_required(VERSION 3.20)
project(xtnetrc_car_runtime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(XTNETRC_BUILD_VISION "Build live/replay runtime using OpenCV" ON)
if(NOT TARGET xtnetrc_hardware)
    add_subdirectory(../hardware hardware_build)
endif()
if(NOT TARGET xtnetrc_motion)
    add_subdirectory(../motion_control motion_build)
endif()
if(NOT TARGET xtnetrc_gps_speed)
    add_subdirectory(../gps_speed_control gps_build)
endif()
add_library(xtnetrc_runtime_core STATIC src/control.cpp)
target_include_directories(xtnetrc_runtime_core PUBLIC include)
target_link_libraries(xtnetrc_runtime_core PUBLIC xtnetrc_hardware xtnetrc_motion xtnetrc_gps_speed)
target_compile_options(xtnetrc_runtime_core PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic>)
include(CTest)
if(BUILD_TESTING)
    add_executable(xtnetrc_runtime_test tests/control_test.cpp)
    target_link_libraries(xtnetrc_runtime_test PRIVATE xtnetrc_runtime_core)
    add_test(NAME xtnetrc_runtime_test COMMAND xtnetrc_runtime_test)
endif()
if(XTNETRC_BUILD_VISION)
    if(NOT TARGET xtnetrc_lane_corridor)
        add_subdirectory(../visual_distance_test perception_build)
    endif()
    add_library(xtnetrc_runtime_app STATIC src/config.cpp src/vision_worker.cpp src/app.cpp)
    target_link_libraries(xtnetrc_runtime_app PUBLIC xtnetrc_runtime_core xtnetrc_lane_corridor xtnetrc_camera_io)
    target_compile_options(xtnetrc_runtime_app PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic>)
    add_executable(xtnetrc_car_runtime src/main.cpp)
    target_link_libraries(xtnetrc_car_runtime PRIVATE xtnetrc_runtime_app)
    add_executable(xtnetrc_lane_follow_test ../visual_distance_test/src/lane_follow_test.cpp)
    target_link_libraries(xtnetrc_lane_follow_test PRIVATE xtnetrc_runtime_app)
    set_target_properties(xtnetrc_car_runtime xtnetrc_lane_follow_test PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    if(BUILD_TESTING)
        add_executable(xtnetrc_runtime_io_test tests/io_test.cpp)
        target_link_libraries(xtnetrc_runtime_io_test PRIVATE xtnetrc_runtime_app)
        target_compile_definitions(xtnetrc_runtime_io_test PRIVATE
            XTNETRC_RUNTIME_CONFIG="${CMAKE_CURRENT_SOURCE_DIR}/../config/runtime_2023.json")
        add_test(NAME xtnetrc_runtime_io_test COMMAND xtnetrc_runtime_io_test)
        set_tests_properties(xtnetrc_runtime_io_test PROPERTIES TIMEOUT 10)
        add_test(NAME xtnetrc_runtime_help COMMAND xtnetrc_car_runtime --help)
        add_test(NAME xtnetrc_runtime_replay_cannot_drive COMMAND xtnetrc_car_runtime
            --replay-image absent.png --enable-motor --i-understand-vehicle-will-move)
        set_tests_properties(xtnetrc_runtime_replay_cannot_drive PROPERTIES WILL_FAIL TRUE)
    endif()
endif()
