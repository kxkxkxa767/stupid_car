cmake_minimum_required(VERSION 3.20)
project(xtnetrc_hardware LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(XTNETRC_WITH_PIGPIO "Build the Raspberry Pi motor output adapter" OFF)
find_package(Threads REQUIRED)
add_library(xtnetrc_hardware STATIC src/vehicle.cpp src/pigpio_driver.cpp src/serial.cpp)
target_include_directories(xtnetrc_hardware PUBLIC include)
target_link_libraries(xtnetrc_hardware PUBLIC Threads::Threads PRIVATE ${CMAKE_DL_LIBS})
if(XTNETRC_WITH_PIGPIO)
    target_compile_definitions(xtnetrc_hardware PRIVATE XTNETRC_WITH_PIGPIO)
endif()
target_compile_options(xtnetrc_hardware PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic>)
include(CTest)
if(BUILD_TESTING)
    add_executable(xtnetrc_hardware_test tests/vehicle_test.cpp)
    target_link_libraries(xtnetrc_hardware_test PRIVATE xtnetrc_hardware)
    add_test(NAME xtnetrc_hardware_test COMMAND xtnetrc_hardware_test)
    set_tests_properties(xtnetrc_hardware_test PROPERTIES TIMEOUT 10)
endif()
