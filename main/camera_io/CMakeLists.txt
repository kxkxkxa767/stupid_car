cmake_minimum_required(VERSION 3.20)
project(xtnetrc_camera_io LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT TARGET xtnetrc_ground_projection)
    add_subdirectory(../ground_projection ground_projection_build)
endif()
find_package(OpenCV 4 REQUIRED COMPONENTS core imgcodecs)
find_package(Threads REQUIRED)
add_library(xtnetrc_camera_io STATIC src/frame_source.cpp src/v4l2_camera.cpp)
target_include_directories(xtnetrc_camera_io PUBLIC include)
target_link_libraries(xtnetrc_camera_io PUBLIC xtnetrc_ground_projection opencv_imgcodecs Threads::Threads)
target_compile_options(xtnetrc_camera_io PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic>)
include(CTest)
if(BUILD_TESTING)
    add_executable(xtnetrc_camera_io_test tests/frame_source_test.cpp)
    target_link_libraries(xtnetrc_camera_io_test PRIVATE xtnetrc_camera_io)
    add_test(NAME xtnetrc_camera_io_test COMMAND xtnetrc_camera_io_test)
endif()
